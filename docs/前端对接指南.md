## API接口列表

### 基础URL
```
http://localhost:8000/api/market
```

### 接口详情

| 接口 | 方法 | 说明 |
|------|------|------|
| `/spot/{code}` | GET | 实时行情（按代码） |
| `/spot/name/{name}` | GET | 实时行情（按名称） |
| `/ranking` | GET | 涨幅榜/跌幅榜 |
| `/kline/{code}` | GET | K线数据（按代码） |
| `/kline/name/{name}` | GET | K线数据（按名称） |

### K线接口参数

| 参数 | 类型 | 默认值 | 说明 |
|------|------|--------|------|
| `period` | string | daily | K线周期：daily/weekly/monthly |
| `start_date` | string | - | 起始日期（2024-01-01） |
| `end_date` | string | - | 结束日期（2024-12-31） |
| `limit` | int | - | 返回数据条数（1-1000） |

### K线图组件

```tsx
// components/KlineChart.tsx
import React, { useEffect, useState } from 'react';
import ReactECharts from 'echarts-for-react';
import axios from 'axios';

interface KlineData {
  date: string;
  open: number;
  close: number;
  high: number;
  low: number;
  volume: number;
  change_pct: number;
}

interface KlineChartProps {
  fundCode: string;
  period?: 'daily' | 'weekly' | 'monthly';
  limit?: number;
}

const KlineChart: React.FC<KlineChartProps> = ({
  fundCode,
  period = 'daily',
  limit = 60
}) => {
  const [data, setData] = useState<KlineData[]>([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    fetchKlineData();
  }, [fundCode, period, limit]);

  const fetchKlineData = async () => {
    try {
      setLoading(true);
      const response = await axios.get(
        `http://localhost:8000/api/market/kline/${fundCode}`,
        { params: { period, limit } }
      );
      setData(response.data.items);
    } catch (error) {
      console.error('获取K线数据失败:', error);
    } finally {
      setLoading(false);
    }
  };

  const getOption = () => {
    const dates = data.map(item => item.date);
    const klineData = data.map(item => [item.open, item.close, item.low, item.high]);
    const volumes = data.map(item => item.volume);

    return {
      title: {
        text: `${fundCode} K线图`,
        left: 'center'
      },
      tooltip: {
        trigger: 'axis',
        axisPointer: { type: 'cross' },
        formatter: function (params: any) {
          const data = params[0].data;
          return `
            日期: ${params[0].name}<br/>
            开盘: ${data[0]}<br/>
            收盘: ${data[1]}<br/>
            最低: ${data[2]}<br/>
            最高: ${data[3]}
          `;
        }
      },
      legend: {
        data: ['K线', '成交量'],
        bottom: 10
      },
      grid: [
        { left: '10%', right: '8%', top: '10%', height: '50%' },
        { left: '10%', right: '8%', top: '70%', height: '20%' }
      ],
      xAxis: [
        {
          type: 'category',
          data: dates,
          gridIndex: 0
        },
        {
          type: 'category',
          data: dates,
          gridIndex: 1
        }
      ],
      yAxis: [
        {
          scale: true,
          gridIndex: 0,
          splitLine: { show: true }
        },
        {
          scale: true,
          gridIndex: 1,
          splitLine: { show: false }
        }
      ],
      dataZoom: [
        {
          type: 'inside',
          xAxisIndex: [0, 1],
          start: 50,
          end: 100
        },
        {
          show: true,
          xAxisIndex: [0, 1],
          type: 'slider',
          bottom: 10,
          start: 50,
          end: 100
        }
      ],
      series: [
        {
          name: 'K线',
          type: 'candlestick',
          data: klineData,
          xAxisIndex: 0,
          yAxisIndex: 0,
          itemStyle: {
            color: '#ef232a',
            color0: '#14b143',
            borderColor: '#ef232a',
            borderColor0: '#14b143'
          }
        },
        {
          name: '成交量',
          type: 'bar',
          data: volumes,
          xAxisIndex: 1,
          yAxisIndex: 1,
          itemStyle: {
            color: function (params: any) {
              const idx = params.dataIndex;
              return data[idx].close >= data[idx].open ? '#ef232a' : '#14b143';
            }
          }
        }
      ]
    };
  };

  if (loading) {
    return <div className="loading">加载中...</div>;
  }

  return (
    <div className="kline-chart">
      <ReactECharts
        option={getOption()}
        style={{ height: '500px', width: '100%' }}
        notMerge={true}
        lazyUpdate={true}
      />
    </div>
  );
};

export default KlineChart;
```

### K线图组件

```tsx
// components/LightweightKline.tsx
import React, { useEffect, useRef, useState } from 'react';
import { createChart, IChartApi, CandlestickData } from 'lightweight-charts';
import axios from 'axios';

interface KlineData {
  date: string;
  open: number;
  close: number;
  high: number;
  low: number;
  volume: number;
}

interface LightweightKlineProps {
  fundCode: string;
  period?: 'daily' | 'weekly' | 'monthly';
  limit?: number;
}

const LightweightKline: React.FC<LightweightKlineProps> = ({
  fundCode,
  period = 'daily',
  limit = 100
}) => {
  const chartRef = useRef<HTMLDivElement>(null);
  const chartInstance = useRef<IChartApi | null>(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    fetchKlineData();

    return () => {
      if (chartInstance.current) {
        chartInstance.current.remove();
      }
    };
  }, [fundCode, period, limit]);

  const fetchKlineData = async () => {
    try {
      setLoading(true);
      const response = await axios.get(
        `http://localhost:8000/api/market/kline/${fundCode}`,
        { params: { period, limit } }
      );

      const klineData: CandlestickData[] = response.data.items.map((item: KlineData) => ({
        time: item.date,
        open: item.open,
        high: item.high,
        low: item.low,
        close: item.close,
      }));

      renderChart(klineData);
    } catch (error) {
      console.error('获取K线数据失败:', error);
    } finally {
      setLoading(false);
    }
  };

  const renderChart = (data: CandlestickData[]) => {
    if (!chartRef.current) return;

    // 清除旧图表
    if (chartInstance.current) {
      chartInstance.current.remove();
    }

    // 创建新图表
    const chart = createChart(chartRef.current, {
      width: chartRef.current.clientWidth,
      height: 500,
      layout: {
        background: { color: '#ffffff' },
        textColor: '#333',
      },
      grid: {
        vertLines: { color: '#e1e1e1' },
        horzLines: { color: '#e1e1e1' },
      },
      crosshair: {
        mode: 1,
      },
      rightPriceScale: {
        borderColor: '#cccccc',
      },
      timeScale: {
        borderColor: '#cccccc',
        timeVisible: true,
        secondsVisible: false,
      },
    });

    const candlestickSeries = chart.addCandlestickSeries({
      upColor: '#ef232a',
      downColor: '#14b143',
      borderUpColor: '#ef232a',
      borderDownColor: '#14b143',
      wickUpColor: '#ef232a',
      wickDownColor: '#14b143',
    });

    candlestickSeries.setData(data);

    chartInstance.current = chart;

    // 响应式调整
    const handleResize = () => {
      if (chartRef.current && chartInstance.current) {
        chartInstance.current.applyOptions({ width: chartRef.current.clientWidth });
      }
    };
    window.addEventListener('resize', handleResize);
  };

  if (loading) {
    return <div className="loading">加载中...</div>;
  }

  return <div ref={chartRef} className="lightweight-kline" />;
};

export default LightweightKline;
```

## 实时行情组件

```tsx
// components/RealtimeQuote.tsx
import React, { useEffect, useState } from 'react';
import axios from 'axios';

interface QuoteData {
  code: string;
  name: string;
  price: number;
  change: number;
  change_pct: number;
  prev_close: number;
  open: number;
  high: number;
  low: number;
  volume: number;
  amount: number;
  turnover_rate: number;
  bid_price: number;
  ask_price: number;
  update_time: string;
}

interface RealtimeQuoteProps {
  fundCode: string;
  autoRefresh?: boolean;
  refreshInterval?: number;
}

const RealtimeQuote: React.FC<RealtimeQuoteProps> = ({
  fundCode,
  autoRefresh = true,
  refreshInterval = 5000
}) => {
  const [data, setData] = useState<QuoteData | null>(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    fetchQuote();

    if (autoRefresh) {
      const timer = setInterval(fetchQuote, refreshInterval);
      return () => clearInterval(timer);
    }
  }, [fundCode, autoRefresh, refreshInterval]);

  const fetchQuote = async () => {
    try {
      const response = await axios.get(
        `http://localhost:8000/api/market/spot/${fundCode}`
      );
      setData(response.data.data);
    } catch (error) {
      console.error('获取行情失败:', error);
    } finally {
      setLoading(false);
    }
  };

  if (loading || !data) {
    return <div className="loading">加载中...</div>;
  }

  const isUp = data.change >= 0;

  return (
    <div className="realtime-quote">
      <div className="quote-header">
        <h2>{data.name}</h2>
        <span className="quote-code">{data.code}</span>
      </div>

      <div className="quote-price">
        <span className={`price ${isUp ? 'up' : 'down'}`}>
          {data.price.toFixed(3)}
        </span>
        <span className={`change ${isUp ? 'up' : 'down'}`}>
          {isUp ? '+' : ''}{data.change.toFixed(3)} ({isUp ? '+' : ''}{data.change_pct.toFixed(2)}%)
        </span>
      </div>

      <div className="quote-details">
        <div className="row">
          <div className="cell">
            <span className="label">今开</span>
            <span className="value">{data.open.toFixed(3)}</span>
          </div>
          <div className="cell">
            <span className="label">昨收</span>
            <span className="value">{data.prev_close.toFixed(3)}</span>
          </div>
        </div>
        <div className="row">
          <div className="cell">
            <span className="label">最高</span>
            <span className="value high">{data.high.toFixed(3)}</span>
          </div>
          <div className="cell">
            <span className="label">最低</span>
            <span className="value low">{data.low.toFixed(3)}</span>
          </div>
        </div>
        <div className="row">
          <div className="cell">
            <span className="label">成交额</span>
            <span className="value">{(data.amount / 1e8).toFixed(2)}亿</span>
          </div>
          <div className="cell">
            <span className="label">换手率</span>
            <span className="value">{data.turnover_rate.toFixed(2)}%</span>
          </div>
        </div>
        <div className="row">
          <div className="cell">
            <span className="label">买一</span>
            <span className="value">{data.bid_price.toFixed(3)}</span>
          </div>
          <div className="cell">
            <span className="label">卖一</span>
            <span className="value">{data.ask_price.toFixed(3)}</span>
          </div>
        </div>
      </div>

      <div className="quote-footer">
        <span className="update-time">更新时间: {data.update_time}</span>
      </div>
    </div>
  );
};

export default RealtimeQuote;
```

## 涨幅榜组件

```tsx
// components/RankingList.tsx
import React, { useEffect, useState } from 'react';
import axios from 'axios';

interface RankingItem {
  code: string;
  name: string;
  price: number;
  change: number;
  change_pct: number;
  turnover_rate: number;
  amount: number;
}

interface RankingListProps {
  type?: 'gain' | 'loss';
  topN?: number;
}

const RankingList: React.FC<RankingListProps> = ({
  type = 'gain',
  topN = 10
}) => {
  const [data, setData] = useState<RankingItem[]>([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    fetchRanking();
  }, [type, topN]);

  const fetchRanking = async () => {
    try {
      setLoading(true);
      const response = await axios.get(
        `http://localhost:8000/api/market/ranking`,
        {
          params: {
            top_n: topN,
            order: type === 'gain' ? 'desc' : 'asc'
          }
        }
      );
      setData(response.data.items);
    } catch (error) {
      console.error('获取榜单失败:', error);
    } finally {
      setLoading(false);
    }
  };

  if (loading) {
    return <div className="loading">加载中...</div>;
  }

  return (
    <div className="ranking-list">
      <div className="ranking-header">
        <h3>{type === 'gain' ? '涨幅榜' : '跌幅榜'}</h3>
        <button onClick={fetchRanking}>刷新</button>
      </div>

      <table>
        <thead>
          <tr>
            <th>排名</th>
            <th>代码</th>
            <th>名称</th>
            <th>最新价</th>
            <th>涨跌幅</th>
            <th>成交额</th>
          </tr>
        </thead>
        <tbody>
          {data.map((item, index) => {
            const isUp = item.change_pct >= 0;
            return (
              <tr key={item.code}>
                <td>{index + 1}</td>
                <td>{item.code}</td>
                <td className="name">{item.name}</td>
                <td>{item.price.toFixed(3)}</td>
                <td className={isUp ? 'up' : 'down'}>
                  {isUp ? '+' : ''}{item.change_pct.toFixed(2)}%
                </td>
                <td>{(item.amount / 1e8).toFixed(2)}亿</td>
              </tr>
            );
          })}
        </tbody>
      </table>
    </div>
  );
};

export default RankingList;
```

## 使用建议

### 图表库选择

| 特性 | ECharts | Lightweight Charts |
|------|---------|-------------------|
| 文件大小 | ~1MB | ~50KB |
| 功能丰富度 | ⭐⭐⭐⭐⭐ | ⭐⭐⭐ |
| 金融专业度 | ⭐⭐⭐ | ⭐⭐⭐⭐⭐ |
| 性能 | 良好 | 优秀 |
| 学习成本 | 中等 | 低 |

**推荐**：
- 追求轻量和金融专业性 → **Lightweight Charts**
- 需要丰富的图表类型和自定义 → **ECharts**

### 数据刷新策略

```typescript
// 自动刷新（5秒）
useEffect(() => {
  const timer = setInterval(fetchQuote, 5000);
  return () => clearInterval(timer);
}, []);

// 节流刷新（仅在页面可见时）
useEffect(() => {
  const handleVisibility = () => {
    if (document.hidden) {
      clearInterval(timer);
    } else {
      fetchQuote();
      timer = setInterval(fetchQuote, 5000);
    }
  };
  document.addEventListener('visibilitychange', handleVisibility);
  return () => document.removeEventListener('visibilitychange', handleVisibility);
}, []);
```
